Comprehensive Async Parallel Dashboard

This vignette provides a comprehensive interactive Shiny application demonstrating all features of the randomwalk package with async parallel processing in WebAssembly.

Complete Showcase: This dashboard demonstrates all randomwalk capabilities:

  • ✅ Async parallel processing (mirai in WebR, crew in native R)
  • ✅ Multiple visualizations (fractal plots, path plots, distributions)
  • ✅ Comprehensive statistics and debugging information
  • ✅ Real-time parameter tuning
  • ✅ Error handling and status updates
  • ✅ Cross-references to documentation

server <- function(input, output, session) {

  # Reactive value to store simulation result
  sim_result <- reactiveVal(NULL)

  # Periodic timer for debug updates
  timer <- reactiveTimer(1000)  # Update every second

  # Run simulation when button clicked
  observeEvent(input$run_sim, {
    output$status <- renderText({
      paste0(
        "🏃 Running simulation...\n",
        "Grid: ", input$grid_size, "×", input$grid_size, "\n",
        "Walkers: ", input$n_walkers, "\n",
        "Workers: ", input$workers,
        if (input$workers > 0) " (async)" else " (sync)", "\n",
        "Started: ", format(Sys.time(), "%H:%M:%S")
      )
    })

    tryCatch({
      start_time <- Sys.time()

      # Run simulation with async support
      result <- run_simulation(
        grid_size = input$grid_size,
        n_walkers = input$n_walkers,
        neighborhood = input$neighborhood,
        boundary = input$boundary,
        max_steps = input$max_steps,
        workers = input$workers,  # Auto-detects backend (mirai in WebR)
        sync_mode = "static",     # Grid snapshots (dynamic is Issue #130)
        verbose = FALSE
      )

      end_time <- Sys.time()
      elapsed <- as.numeric(difftime(end_time, start_time, units = "secs"))

      sim_result(result)

      output$status <- renderText({
        paste0(
          "✅ Simulation complete!\n",
          "Elapsed: ", sprintf("%.2f", elapsed), "s\n",
          "Backend: ", if (input$workers > 0) {
            if (is_webr()) "mirai (WebR)" else "crew (native R)"
          } else {
            "synchronous"
          }, "\n",
          "Completed: ", format(end_time, "%H:%M:%S")
        )
      })

    }, error = function(e) {
      output$status <- renderText({
        paste0(
          "❌ ERROR:\n",
          e$message, "\n\n",
          "Troubleshooting:\n",
          "- Check JavaScript console (F12)\n",
          "- Verify randomwalk package loaded\n",
          "- Check workers setting (0-4)\n",
          "- See WIKI_SHINYLIVE_LESSONS_LEARNED.md"
        )
      })
    })
  })

  # PAGE 1: Fractal Graph
  output$fractal_plot <- renderPlot({
    req(sim_result())
    plot_grid(sim_result(), main = "Fractal Pattern - Black Pixels on Grid")
  })

  output$fractal_stats <- renderText({
    req(sim_result())
    stats <- sim_result()$statistics
    paste0(
      "Black Pixels: ", stats$black_pixels,
      " (", sprintf("%.2f%%", stats$black_percentage), " of grid)\n",
      "Grid Size: ", stats$grid_size, "×", stats$grid_size,
      " (", stats$grid_size^2, " total cells)"
    )
  })

  # PAGE 2: Walker Paths
  output$paths_plot_first25 <- renderPlot({
    req(sim_result())
    result <- sim_result()
    n_walkers <- length(result$walkers)

    if (n_walkers >= 1) {
      first_n <- min(25, n_walkers)
      plot_walker_paths(result, walker_ids = 1:first_n,
                        main = paste("First", first_n, "Walkers"))
    } else {
      plot.new()
      text(0.5, 0.5, "No walkers available")
    }
  })

  output$paths_plot_last25 <- renderPlot({
    req(sim_result())
    result <- sim_result()
    n_walkers <- length(result$walkers)

    if (n_walkers > 25) {
      last_n <- min(25, n_walkers)
      start_id <- n_walkers - last_n + 1
      plot_walker_paths(result, walker_ids = start_id:n_walkers,
                        main = paste("Last", last_n, "Walkers"))
    } else {
      plot.new()
      text(0.5, 0.5, "Less than 25 walkers total - see first panel")
    }
  })

  # PAGE 3: Distributions
  output$dist_overall <- renderPlot({
    req(sim_result())
    result <- sim_result()

    # Extract path lengths
    path_lengths <- sapply(result$walkers, function(w) nrow(w$path))

    # Plot histogram
    hist(path_lengths,
         breaks = 30,
         col = "steelblue",
         border = "white",
         main = "Overall Path Length Distribution",
         xlab = "Path Length (steps)",
         ylab = "Frequency")
    abline(v = median(path_lengths), col = "red", lwd = 2, lty = 2)
    legend("topright",
           legend = c(paste("Median:", median(path_lengths)),
                     paste("Mean:", sprintf("%.1f", mean(path_lengths)))),
           col = c("red", "black"),
           lty = c(2, 1),
           lwd = 2)
  })

  output$dist_by_reason <- renderPlot({
    req(sim_result())
    result <- sim_result()

    # Extract termination reasons and path lengths
    walkers_data <- data.frame(
      path_length = sapply(result$walkers, function(w) nrow(w$path)),
      reason = sapply(result$walkers, function(w) w$termination_reason)
    )

    # Plot boxplot by termination reason
    ggplot(walkers_data, aes(x = reason, y = path_length, fill = reason)) +
      geom_boxplot(alpha = 0.7) +
      geom_jitter(width = 0.2, alpha = 0.3) +
      labs(title = "Path Lengths by Termination Reason",
           x = "Termination Reason",
           y = "Path Length (steps)") +
      theme_minimal() +
      theme(legend.position = "none")
  })

  # PAGE 4: Statistics
  output$stats_comprehensive <- renderText({
    req(sim_result())
    stats <- sim_result()$statistics

    paste0(
      "=== GRID STATISTICS ===\n",
      "Grid Size: ", stats$grid_size, "×", stats$grid_size, "\n",
      "Total Cells: ", stats$grid_size^2, "\n",
      "Black Pixels: ", stats$black_pixels,
      sprintf(" (%.2f%%)\n", stats$black_percentage),
      "White Pixels: ", stats$grid_size^2 - stats$black_pixels,
      sprintf(" (%.2f%%)\n", 100 - stats$black_percentage),
      "\n=== WALKER STATISTICS ===\n",
      "Total Walkers: ", stats$total_walkers, "\n",
      "Completed: ", stats$completed_walkers, "\n",
      "Success Rate: ",
      sprintf("%.1f%%", 100 * stats$completed_walkers / stats$total_walkers), "\n",
      "\n=== STEP STATISTICS ===\n",
      "Total Steps: ", stats$total_steps, "\n",
      "Min Steps: ", stats$min_steps, "\n",
      "Max Steps: ", stats$max_steps, "\n",
      "Mean Steps: ", sprintf("%.1f", stats$mean_steps), "\n",
      "Median Steps: ", stats$median_steps, "\n",
      "Std Dev: ", sprintf("%.1f", sd(sapply(sim_result()$walkers, function(w) nrow(w$path)))), "\n",
      "\n=== PERFORMANCE ===\n",
      "Elapsed Time: ", sprintf("%.2f", stats$elapsed_time_secs), " seconds\n",
      "Steps per Second: ",
      sprintf("%.0f", stats$total_steps / stats$elapsed_time_secs), "\n",
      "\n=== TERMINATION REASONS ===\n",
      paste(sprintf("  %s: %d (%.1f%%)",
                    names(stats$termination_reasons),
                    stats$termination_reasons,
                    100 * stats$termination_reasons / stats$total_walkers),
            collapse = "\n")
    )
  })

  # PAGE 5: Debug - Versions
  output$debug_versions <- renderText({
    paste0(
      "=== PACKAGE VERSIONS ===\n",
      "randomwalk: ", packageVersion("randomwalk"), "\n",
      "shiny: ", packageVersion("shiny"), "\n",
      "ggplot2: ", packageVersion("ggplot2"), "\n",
      if (requireNamespace("mirai", quietly = TRUE)) {
        paste0("mirai: ", packageVersion("mirai"), "\n")
      } else "",
      if (requireNamespace("nanonext", quietly = TRUE)) {
        paste0("nanonext: ", packageVersion("nanonext"), "\n")
      } else "",
      if (requireNamespace("crew", quietly = TRUE)) {
        paste0("crew: ", packageVersion("crew"), "\n")
      } else ""
    )
  })

  # PAGE 5: Debug - Inputs
  output$debug_inputs <- renderText({
    paste0(
      "=== CURRENT INPUTS ===\n",
      "Grid Size: ", input$grid_size, "\n",
      "Number of Walkers: ", input$n_walkers, "\n",
      "Parallel Workers: ", input$workers, "\n",
      "Neighborhood: ", input$neighborhood, "\n",
      "Boundary: ", input$boundary, "\n",
      "Max Steps: ", input$max_steps
    )
  })

  # PAGE 5: Debug - Backend
  output$debug_backend <- renderText({
    env_is_webr <- is_webr()

    paste0(
      "=== BACKEND INFORMATION ===\n",
      "Environment: ", if (env_is_webr) "WebR/Browser" else "Native R", "\n",
      "R Version: ", R.version.string, "\n",
      "Backend: ", if (input$workers > 0) {
        if (env_is_webr) "mirai (WebAssembly-compatible)" else "crew"
      } else {
        "synchronous (no parallelization)"
      }, "\n",
      "Sync Mode: static (grid snapshots)\n",
      "\n=== ENVIRONMENT DETECTION ===\n",
      ".webr_env exists: ", exists(".webr_env", envir = .GlobalEnv), "\n",
      "WEBR env var: ", Sys.getenv("WEBR"), "\n",
      "is_webr() result: ", env_is_webr
    )
  })

  # PAGE 5: Debug - Periodic Updates
  output$debug_periodic <- renderText({
    # Depend on timer to update every second
    timer()

    paste0(
      "=== PERIODIC UPDATES ===\n",
      "Current Time: ", format(Sys.time(), "%Y-%m-%d %H:%M:%S"), "\n",
      "Simulation Result: ", if (is.null(sim_result())) "Not yet run" else "Available", "\n",
      "Active Tab: ", input$tabs
    )
  })
}

           identical(Sys.getenv("WEBR"), "1")

# Backend selection in run_simulation()
if (workers > 0) {
  if (is_webr) {
    # Use mirai backend in WebAssembly
    controller <- create_mirai_controller(workers, seconds_idle)
  } else {
    # Use crew backend in native R
    controller <- crew::crew_controller_local(workers, seconds_idle)
  }
}
